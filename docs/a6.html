// creates variables for the serial connection and reader
let port, reader;

// creates a variable that stores the latest full line received from Arduino
let latestLine = "";

// creates a variable to store the button state from Arduino (1 not pressed, 0 pressed with INPUT_PULLUP)
let buttonState = 1;

// creates a variable to store the potentiometer value from Arduino (0–1023)
let potValue = 0;

// creates a variable to store the LED brightness we send to Arduino (0–255)
let brightness = 0;

// creates a variable to store the connect button element
let connectBtn;


// setup() runs once when the webpage starts
function setup() {

  // creates a canvas for drawing visuals
  createCanvas(500, 350);

  // sets the font for readable debug text
  textFont("monospace");

  // creates a button the user clicks to connect the Arduino
  connectBtn = createButton("Connect Arduino");

  // positions the button near the top-left of the page
  connectBtn.position(10, 10);

  // when the button is pressed, runs the serial connection function
  connectBtn.mousePressed(connectToSerial);

}


// draw() runs repeatedly
function draw() {

  // sets a dark background
  background(20);

  // sets text color to white
  fill(255);

  // sets the text size
  textSize(14);

  // shows the current button state coming from Arduino
  text(`buttonState (1=not pressed, 0=pressed): ${buttonState}`, 10, 70);

  // shows the current potentiometer value coming from Arduino
  text(`potValue (0–1023): ${potValue}`, 10, 90);

  // shows the brightness value we are sending to Arduino
  text(`brightness sent (0–255): ${brightness}`, 10, 110);

  // shows the raw last line received from Arduino for debugging
  text(`raw from Arduino: ${latestLine}`, 10, 130);

  // maps the pot value to a circle size
  let r = map(potValue, 0, 1023, 10, 200);

  // if the Arduino button is pressed, show the circle
  if (buttonState === 0) {

    // set the fill based on brightness for a visible link between web input and drawing
    fill(brightness);

    // remove the circle outline
    noStroke();

    // draws the circle
    circle(width * 0.65, height * 0.55, r);

  } else {

    // show instructions when the button is not pressed
    fill(150);

    // write instruction text
    text("Press the Arduino button to show the circle", 10, 160);

  }

}


// mouseMoved() runs whenever the mouse moves on the page
async function mouseMoved() {

  // stops if the serial port is not connected yet
  if (!port) return;

  // maps the mouse x position to LED brightness range (0–255)
  brightness = map(mouseX, 0, width, 0, 255);

  // constrains brightness to valid bounds and round to an integer
  brightness = Math.round(constrain(brightness, 0, 255));

  // gets a writer so data is sent to the Arduino
  const writer = port.writable.getWriter();

  // sends brightness with a newline so Arduino Serial.parseInt() reads it cleanly
  await writer.write(new TextEncoder().encode(brightness + "\n"));

  // releases the writer lock so it can be used again later
  writer.releaseLock();

}


// connectToSerial() opens the WebSerial connection
async function connectToSerial() {

  try {

    // asks the user to select the Arduino serial port
    port = await navigator.serial.requestPort();

    // opens the port at the same baud rate used in Arduino code
    await port.open({ baudRate: 9600 });

    // creates a text decoder stream to convert incoming bytes to text
    const decoder = new TextDecoderStream();

    // pipes serial data through the decoder
    port.readable.pipeTo(decoder.writable);

    // gets a reader to read the decoded text stream
    reader = decoder.readable.getReader();

    // starts the continuous serial reading loop
    readSerialLoop();

  } catch (err) {

    // logs any errors that happen during connection
    console.log("Serial connection error:", err);

  }

}


// readSerialLoop() continuously reads lines from Arduino
async function readSerialLoop() {

  // creates a buffer string to store partial serial data
  let buffer = "";

  // keeps reading forever
  while (true) {

    // reads the next chunk of decoded text from serial
    const { value, done } = await reader.read();

    // stops the loop if the stream is closed
    if (done) break;

    // skips if no value was received
    if (!value) continue;

    // adds the new chunk to the buffer
    buffer += value;

    // splits the buffer into lines using newline characters
    let lines = buffer.split("\n");

    // keeps the last partial line in the buffer
    buffer = lines.pop();

    // processes each complete line
    for (let line of lines) {

      // removes extra spaces from the line
      line = line.trim();

      // skips empty lines
      if (!line) continue;

      // stores the latest raw line for display/debugging
      latestLine = line;

      // splits the CSV line into parts
      const parts = line.split(",");

      // Arduino sends: potValue,buttonState
      if (parts.length === 2) {

        // converts pot value from string to number
        potValue = Number(parts[0]);

        // converts button state from string to number
        buttonState = Number(parts[1]);

      }

    }

  }

}

Circuit's Operation

Context:
Inside the webpage, a button prompts the user to connect the Arduino. Once the user presses the button on the webpage,
the user can then press the button on the bread board to turn the LED on. Once that is on,
the LED fades based on the location of the mouse on the screen on the X-axis.
The mouse's location also determines the opacity of the circle displayed on the webpage.
The potentiometer determines the size of the circle displayed on the webpage.
Additionally, a display showing the button state, the potentiometer value, the LED brightness, and the serial reading from the Arduino is shown.
